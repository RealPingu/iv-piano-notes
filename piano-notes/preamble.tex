% ==============================================================================
% 1. Page Layout & Spacing
% ==============================================================================
\usepackage[utf8]{inputenc}
\usepackage[margin=1in]{geometry}       % 1 inch margins on all sides
\usepackage{parskip}                    % Paragraph spacing instead of indentation

% ==============================================================================
% 2. Math & Typography
% ==============================================================================
\usepackage{amsmath, amssymb, amsthm}   % Standard math symbols and environments
\usepackage{microtype}                  % Improved typography & spacing

% ==============================================================================
% 3. Lists, Tables & Graphics
% ==============================================================================
\usepackage{enumitem}                   % Compact list spacing
\setlist{nosep}
\usepackage{booktabs}                   % Professional tables (\toprule, \midrule, \bottomrule)
\usepackage{graphicx}                   % Figures & images
\graphicspath{{figures/}{images/}}      % Auto-search folders for \includegraphics

% ==============================================================================
% 4. Callout Boxes (tcolorbox)
% ==============================================================================
\usepackage[most]{tcolorbox}

% Definition Box (Blue)
\newtcolorbox{definitionbox}[1]{
    colback=blue!5!white,
    colframe=blue!75!black,
    fonttitle=\bfseries,
    title=#1,
    arc=2mm
}

% Warning / Important Note Box (Orange/Red)
\newtcolorbox{warningbox}[1]{
    colback=orange!5!white,
    colframe=orange!80!black,
    fonttitle=\bfseries,
    title=#1,
    arc=2mm
}

% Example / Tip Box (Green)
\newtcolorbox{examplebox}[1]{
    colback=green!5!white,
    colframe=green!60!black,
    fonttitle=\bfseries,
    title=#1,
    arc=2mm
}

% ==============================================================================
% 5. Hyperlinks & PDF Metadata (Always loaded last)
% ==============================================================================
\usepackage{hyperref}
\hypersetup{
    colorlinks=true,
    linkcolor=blue!80!black,
    urlcolor=blue!70!black,
    citecolor=green!60!black
}

% ==============================================================================
% 6. Musical Symbols & Unicode Characters
% ==============================================================================
\usepackage{newunicodechar}
\newunicodechar{♭}{\texorpdfstring{\ensuremath{\flat}}{b}}
\newunicodechar{♯}{\texorpdfstring{\ensuremath{\sharp}}{\#}}
\newunicodechar{♮}{\texorpdfstring{\ensuremath{\natural}}{}}
